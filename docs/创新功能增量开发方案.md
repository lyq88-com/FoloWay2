# 独响 APP · 8+3 差异化创新功能增量开发方案

> 版本：v1.2 ｜ 日期：2026-09-21
> 开发原则：**全部增量开发，不推翻原有架构**，复用现有库表 / 接口 / 页面 / 智谱 glm-4-flash 调用链路。
> v1.1 增量：追加「功能 9 成长日历」「功能 10 个人设置页优化」两大功能（共 4 个子能力），见第 6 章。
> v1.2 增量：追加「功能 11 MBTI + 大五人格 专业测评」（5 张新表 + 1 套专业题库 + 综合报告 + AI 人设深度联动），见第 7 章；原有 4 类人格体系（thinker/feeler/doer/dreamer）**保留为 AI 人设兼容层**，由 MBTI 推导映射，不破坏既有 AI 对话/树洞/正念推荐逻辑。

## 0. 开工前必读：现有底座盘点（决定「改」还是「建」）

在动手前，先对齐一件重要事实：本项目实际架构为 **FastAPI（8000 业务后端）+ Flask（5000 鉴权 / 5001 AI 独立服务）+ uni-app Vue3 前端**，模型层使用 SQLAlchemy（`server/app/models/`）。下文所有字段名、表名、接口路径均**逐一核对过真实源码**，可直接落地。

### 1.9 三个可选轻量功能

**① 免愧疚暂停模式**
- `habits` 加：`paused_from DATE NULL`, `paused_until DATE NULL`, `pause_reason VARCHAR(200) DEFAULT ''`

```python
    @staticmethod
    def _calc_features(db, user_id, today) -> dict:
        """读取行为数据 → 人格判定特征（纯只读，不改动任何现有业务表）"""
        d7 = today - timedelta(days=6)

        # 1) 连续摆烂天数：从 checkin_items.status='slacking' 倒着数
        slack_days = [
            r.created_at.date() for r in db.query(CheckinItem).filter(
                CheckinItem.user_id == user_id,
                CheckinItem.status == "slacking",
                CheckinItem.created_at >= d7,
            ).all()
        ]
        slack_streak = _consecutive_days(set(slack_days), today)

        # 2) 近 7 天低落情绪天数：复用现有 emotion_records
        NEGATIVE = ("sad", "angry", "tired")
        low_mood_days = db.query(EmotionRecord).filter(
            EmotionRecord.user_id == user_id,
            EmotionRecord.record_date >= d7,
            EmotionRecord.emotion.in_(NEGATIVE),
            EmotionRecord.intensity >= 6,      # 强度够高才算
        ).distinct(EmotionRecord.record_date).count()

        # 3) 连续完成天数
        done_streak = db.query(Checkin).filter(
            Checkin.user_id == user_id, Checkin.checkin_date == today
        ).first()
        done_streak = done_streak.streak_days if done_streak else 0

        # 4) 危机优先（对接现有 crisis_monitor_service，最高优先级）
        risk_level = 0
        try:
            risk_level = CrisisService.current_risk_level(db, user_id)
        except Exception:
            pass

        return dict(slack_streak=slack_streak, low_mood_days=low_mood_days,
                    done_streak=done_streak, risk_level=risk_level)

    @staticmethod
    def evaluate(db, user_id, force=False):
        """人格评估 —— 纯规则决策，确定性、可解释、零成本，不调大模型"""
        today = date.today()
        f = PersonaService._calc_features(db, user_id, today)

        # === 决策优先级（自上而下，命中即停）===
        # P0 危机：交给现有 crisis 模块，不抢人格
        if f["risk_level"] >= 3:
            new_persona = "companion"
        # P1 情绪优先：低落就陪伴，绝不在人难受时督促（心理健康伦理红线）
        elif f["low_mood_days"] >= 3:
            new_persona = "companion"
        # P2 连续摆烂 ≥ 3 天 → 清醒督促
        elif f["slack_streak"] >= 3:
            new_persona = "coach"
        # P3 刚刚完成打卡 / 连续达标 → 鼓励玩伴
        elif f["done_streak"] >= 3:
            new_persona = "playmate"
        else:
            new_persona = "companion_gentle"

        state = db.query(AIPersonaState).filter_by(user_id=user_id).first()
        if state is None:
            state = AIPersonaState(user_id=user_id, persona=new_persona, **f)
            db.add(state)
            db.flush()
        elif state.persona != new_persona or force:
            db.add(AIPersonaLog(
                user_id=user_id, from_persona=state.persona, to_persona=new_persona,
                trigger_type=PersonaService._trigger_of(f),
                evidence=json.dumps(f, ensure_ascii=False),
            ))
            state.persona = new_persona
        state.slack_streak, state.low_mood_days = f["slack_streak"], f["low_mood_days"]
        state.done_streak, state.risk_level = f["done_streak"], f["risk_level"]
        state.evidence = json.dumps(f, ensure_ascii=False)
        state.evaluated_at = datetime.now()
        return state

            or_(AIUserMemory.expire_at.is_(None), AIUserMemory.expire_at >= today),
        ).order_by(AIUserMemory.weight.desc(), AIUserMemory.created_at.desc()) \
         .limit(AnchorService.MAX_ANCHORS).all()

            tag = f"（{'、'.join(json.loads(m.tags or '[]'))}）" if m.tags else ""
            lines.append(f"- {m.memory_content}{tag}")

        tags = db.query(UserBehaviorTag).filter_by(user_id=user_id) \
                 .order_by(UserBehaviorTag.weight.desc()).limit(AnchorService.MAX_TAGS).all()

            lines.append("- 这位用户反复出现的模式：" +
                         "、".join(f"{t.tag}({t.weight}次)" for t in tags))
```

### 3.2 分层耦合 MDP 自适应目标引擎（功能 4）

> **核心设计**：现有 `difficulty_logs` 已经是「7 天完成率阈值触发升降级」的**规则版**。本方案在其上**套一层 MDP 决策**：规则版继续作为「安全护栏」，MDP 负责在护栏内选择最优动作。这样既创新又保证不失控。

```python
# server/app/services/mdp_engine.py

GAMMA = 0.9          # 折扣因子
ALPHA = 0.15         # 学习率
STATE_BINS = ("S0", "S1", "S2", "S3")
ACTIONS = ("down", "keep", "up")

# 先验 Q 表（避免冷启动乱动）：越接近当前状态的动作先验越高
PRIOR_Q = {
    "S0": {"down": 0.7, "keep": 0.2, "up": -0.5},
    "S1": {"down": 0.4, "keep": 0.4, "up": 0.0},
    "S2": {"down": 0.0, "keep": 0.4, "up": 0.4},
    "S3": {"down": -0.5, "keep": 0.2, "up": 0.7},
}


class MDPEngine:
    """单习惯马尔可夫决策过程
    状态 S：参与度（由近 7 天完成率 + 摆烂次数映射）
    动作 A：降低 / 保持 / 提升 目标难度
    奖励 R：完成 +1，摆烂 -1，暂停 0，并叠加"难度错配惩罚"
    """

    # ---------- 状态映射 ----------
    @staticmethod
    def to_state(rate7: float, slack7: int, done_streak: int) -> str:
        if rate7 < 0.30 or slack7 >= 3:
            return "S0"      # 低迷
        if rate7 < 0.60:
            return "S1"      # 挣扎
        if rate7 < 0.85 or done_streak < 7:
            return "S2"      # 稳定
        return "S3"          # 精通

    # ---------- 奖励函数 ----------
    @staticmethod
    def reward(completed: bool, status: str, difficulty_rating: int,
               action_taken: str) -> float:
        if status == "paused":
            return 0.0                       # 暂停不奖不罚（免愧疚原则）
        r = 1.0 if status == "done" else -1.0
        # 难度错配惩罚：评级"困难"却完成了 → 说明该升级；评级"简单"却摆烂 → 说明目标不清晰
        if status == "done" and difficulty_rating == 3 and action_taken != "up":
            r -= 0.3
        if status == "slacking" and action_taken == "up":
            r -= 0.5                         # 升级后仍摆烂，重罚该动作
        if status == "slacking" and action_taken == "down":
            r += 0.2                         # 及时降级是正确决策（保护留存）
        return r

    # ---------- Q 更新 ----------
    @staticmethod
    def q_update(qtable: dict, s: str, a: str, r: float, s_next: str) -> float:
        best_next = max(qtable[s_next].values())
        old = qtable[s][a]
        new = old + ALPHA * (r + GAMMA * best_next - old)
        qtable[s][a] = round(new, 3)
        return new

    # ---------- 策略（护栏 + 冷却） ----------
    @staticmethod
    def policy(qtable: dict, s: str, ctx: dict) -> tuple:
        """返回 (action, q_value, reason)"""

        # == 护栏 1：连续摆烂 ≥ 3 → 强制降级，无视 Q 表（防挫败放弃）==
        if ctx["slack_streak"] >= 3 and ctx["current_diff"] != "easy":
            return "down", qtable[s]["down"], f"连续 {ctx['slack_streak']} 天摆烂，先把目标降到你做得到"

        # == 护栏 2：连续完成 ≥ 3 才允许升级（防刚恢复就加压）==
        if ctx["done_streak"] < 3 and ctx["current_diff"] != "hard":
            if max(qtable[s], key=qtable[s].get) == "up":
                return "keep", qtable[s]["keep"], "状态在回升，先稳住这个难度"

        # == 护栏 3：已是最低/最高难度，边界夹紧 ==
        if ctx["current_diff"] == "easy" and max(qtable[s], key=qtable[s].get) == "down":
            return "keep", qtable[s]["keep"], "已是最小目标，保持不变就好"

        action = max(qtable[s], key=qtable[s].get)
        reason_map = {
            "up":   f"近 7 天完成率 {ctx['rate7']:.0%}，可以试着往上走一小步",
            "down": f"近 7 天完成率 {ctx['rate7']:.0%}，先把目标调小一点，别让挫败感劝退你",
            "keep": "当前难度和目标匹配得正好，继续保持",
        }
        return action, qtable[s][action], reason_map[action]

    # ---------- 主入口 ----------
    @staticmethod
    def adapt(db, user_id: int, habit: Habit) -> dict:
        today = date.today()

        # 冷却检查：同一习惯 7 天内最多调整 1 次
        if habit.adjust_cooldown and habit.adjust_cooldown > today:
            return {"changed": False, "reason": "cooling_down"}

        # 1) 特征计算
        d7 = today - timedelta(days=6)
        items = db.query(CheckinItem).filter(
            CheckinItem.user_id == user_id,
            CheckinItem.habit_id == habit.id,
            CheckinItem.created_at >= d7,
        ).all()
        done = sum(1 for i in items if i.status == "done")
        slack = sum(1 for i in items if i.status == "slacking")
        rate7 = done / max(len(items), 1)
        slack_streak = _consecutive_slack_days(db, user_id, habit.id, today)
        done_streak = habit.continuous_day

        s = MDPEngine.to_state(rate7, slack, done_streak)

        # 2) 取出/初始化 Q 表（存 difficulty_logs.context 或独立 JSON 列，方案见 §1.4）
        qtable = load_qtable(db, user_id, habit.id) or copy.deepcopy(PRIOR_Q)

        # 3) 上一步的奖励回填 + Q 更新
        last = db.query(DifficultyLog).filter_by(user_id=user_id, habit_id=habit.id) \
                 .order_by(DifficultyLog.id.desc()).first()
        if last:
            r = MDPEngine.reward(
                completed=(last.change_type != "slack"),
                status="done" if habit.continuous_day > 0 else "slacking",
                difficulty_rating=items[-1].difficulty_rating if items else 0,
                action_taken=last.change_type,
            )
            MDPEngine.q_update(qtable, last.to_state or s, last.change_type or "keep", r, s)

        # 4) 策略选择
        ctx = dict(rate7=rate7, slack_streak=slack_streak, done_streak=done_streak,
                   current_diff=habit.difficulty)
        action, q_value, reason = MDPEngine.policy(qtable, s, ctx)

        if action == "keep":
            save_qtable(db, user_id, habit.id, qtable)
            return {"changed": False, "reason": reason, "state": s}

        # 5) 目标量映射：复用 task_templates.difficulty_targets JSON（已有字段！）
        new_diff = _shift_difficulty(habit.difficulty, action)     # easy/medium/hard
        tpl = db.query(TaskTemplate).filter_by(id=habit.template_id).first()
        targets = json.loads(tpl.difficulty_targets or "{}") if tpl else {}
        new_target = targets.get(new_diff) or habit.target

        # 6) 幅度限制：目标数值变化不超过 25%（防剧烈跳动）
        new_target = _clamp_target_change(habit.target, new_target, max_ratio=0.25)

        # 7) 落库 + 写日志（复用现有 difficulty_logs 表结构）
        db.add(DifficultyLog(
            user_id=user_id, habit_id=habit.id, habit_name=habit.name,
            old_difficulty=habit.difficulty, new_difficulty=new_diff,
            change_type=action, old_target=habit.target, new_target=new_target,
            recent_rate=int(rate7 * 100), change_date=today,
            trigger_reason="mdp_policy", mdp_q_value=q_value,
            from_state=last.to_state if last else s, to_state=s,
        ))
        db.add(HabitRecommendation(
            user_id=user_id, habit_id=habit.id, rec_difficulty=new_diff,
            rec_target=new_target, rec_level=action, reason=reason,
            expected_rate=_predict_rate(rate7, action), rec_date=today,
        ))
        habit.difficulty, habit.target = new_diff, new_target
        habit.mdp_state, habit.mdp_updated_at = s, datetime.now()
        habit.adjust_cooldown = today + timedelta(days=7)   # 冷却 7 天
        save_qtable(db, user_id, habit.id, qtable)

        # 8) 记忆锚点 + 人格复评（联动）
        AnchorService.record(db, user_id, memory_type="plan",
                             content=f"「{habit.name}」目标{'上调' if action=='up' else '下调'}为 {new_target}",
                             tags=["目标调整"], weight=6)
        PersonaService.evaluate(db, user_id)

        return {"changed": True, "action": action, "old": habit.target,
                "new": new_target, "reason": reason, "q_value": q_value, "state": s}
```

```python
class TreasureService:

    # ---------- 条件判定 DSL ----------
    @staticmethod
    def _eval_condition(cond: dict, ctx: dict, stats: dict) -> bool:
        """cond 示例：
        {"type":"cumulative","key":"checkin_count","value":21}        累计完成 21 次
        {"type":"cumulative","key":"slack_count","value":5}           累计记录 5 次摆烂
        {"type":"streak","habit_type":"sleep","value":7}              某类习惯连续 7 天
        {"type":"emotion_repeat","emotion":"sad","days":3}            连续 3 天低落
        {"type":"slack_with_emotion","emotion":["sad","tired"],"value":3}
        {"type":"all_habits_done","value":1}                          当日全部完成
        {"type":"comeback","value":1}                                 摆烂 3 天后重新完成
        {"type":"habit_combo","codes":["habit_night_owl","habit_early_bird"]}
        """
        t = cond.get("type")
        v = cond.get("value", 1)

        if t == "cumulative":
            return stats.get(cond["key"], 0) >= v
        if t == "streak":
            return stats.get(f"streak_{cond.get('habit_type', 'any')}", 0) >= v
        if t == "emotion_repeat":
            return stats.get(f"low_mood_days_{cond['emotion']}", 0) >= cond["days"]
        if t == "slack_with_emotion":
            return stats.get("slack_days_with_negative_mood", 0) >= v
        if t == "all_habits_done":
            return stats.get("today_all_done", 0) >= v
        if t == "comeback":
            return stats.get("comeback_today", 0) >= v
        if t == "habit_combo":
            return all(c in stats.get("unlocked_codes", []) for c in cond["codes"])
        return False

    # ---------- 解锁入口 ----------
    @staticmethod
    def check_unlock(db, user_id, trigger: str, ctx: dict) -> list:
        """trigger: checkin / slack / emotion / streak / milestone / combo
        在打卡、摆烂、情绪记录成功后调用"""

        # 1) 计算用户当前统计快照（一次算好，供所有候选宝物判定）
        stats = TreasureService._build_stats(db, user_id)
        stats["unlocked_codes"] = [
            d.code for d in db.query(TreasureDef).join(
                UserTreasure, UserTreasure.treasure_id == TreasureDef.id
            ).filter(UserTreasure.user_id == user_id).all()
        ]

        # 2) 候选：条件里 trigger_type 匹配、且尚未解锁
        owned = {ut.treasure_id for ut in
                 db.query(UserTreasure).filter_by(user_id=user_id).all()}
        candidates = db.query(TreasureDef).filter(
            TreasureDef.status == 1,
            TreasureDef.source_type.in_(
                ["emotion"] if trigger in ("slack", "emotion") else ["growth", "emotion"]
            ),
        ).all()

        unlocked = []
        for t in candidates:
            if t.id in owned:
                continue
            cond = json.loads(t.unlock_condition or "{}")
            if cond.get("trigger_type") and cond["trigger_type"] != trigger:
                continue
            if TreasureService._eval_condition(cond, ctx, stats):
                ut = UserTreasure(
                    user_id=user_id, treasure_id=t.id, source_type=t.source_type,
                    trigger_ref_type=trigger, trigger_ref_id=ctx.get("item_id", 0),
                    is_new=1,
                )
                db.add(ut)
                # 解锁即写入「专属对话片段」到记忆锚点，下轮对话 AI 会主动提起
                AnchorService.record(
                    db, user_id, memory_type="achievement",
                    content=f"解锁宝物「{t.name}」：{t.story_text or ''}",
                    tags=["宝物", t.name], weight=8, is_long_term=1,
                )
                unlocked.append({"code": t.code, "name": t.name,
                                 "icon": t.icon, "rarity": t.rarity,
                                 "source_type": t.source_type,
                                 "dialogue_snippet": t.dialogue_snippet})
        db.flush()

        # 3) 组合检查（宝物组合触发专属剧情）
        combos = TreasureService._check_combos(db, user_id, owned | {u["code"] for u in []})
        return {"unlocked": unlocked, "combos": combos}

    # ---------- 组合触发 ----------
    @staticmethod
    def _check_combos(db, user_id, owned_ids) -> list:
        owned_codes = {
            d.code for d in db.query(TreasureDef).join(
                UserTreasure, UserTreasure.treasure_id == TreasureDef.id
            ).filter(UserTreasure.user_id == user_id).all()
        }
        got = []
        already = {uc.combo_id for uc in
                   db.query(UserTreasureCombo).filter_by(user_id=user_id).all()}

        for c in db.query(TreasureComboDef).all():
            if c.id in already:
                continue
            need = json.loads(c.need_codes or "[]")
            if all(code in owned_codes for code in need):
                db.add(UserTreasureCombo(user_id=user_id, combo_id=c.id))

                # 关键：组合解锁 → 开一个专属剧情会话，AI 用 combo 的开场白主动说话
                session = ChatSession(
                    user_id=user_id, role_type="companion",
                    title=c.story_title, session_type="chat",
                    dialogue_style="gentle",
                    context_summary=f"【专属剧情】{c.story_content}",
                )
                db.add(session)
                db.flush()
                db.add(ChatMessage(
                    session_id=session.id, user_id=user_id, role="assistant",
                    content=c.ai_opening, message_type="achievement",
                    is_key_memory=1,
                ))
                got.append({"code": c.code, "name": c.name,
                            "story_title": c.story_title,
                            "session_id": session.id})
        return got
```

### 3.5 因果分析（功能 8）

```python
# server/app/services/causality_service.py

def pearson(x, y):
    n = len(x)
    if n < 7: return 0.0
    mx, my = sum(x)/n, sum(y)/n
    num = sum((a-mx)*(b-my) for a, b in zip(x, y))
    den = (sum((a-mx)**2 for a in x) * sum((b-my)**2 for b in y)) ** 0.5
    return round(num/den, 3) if den else 0.0


def analyze(db, user_id, period="30d"):
    days = {"7d": 7, "30d": 30, "90d": 90}[period]
    dates = [date.today() - timedelta(days=i) for i in range(days-1, -1, -1)]
    idx = {d: i for i, d in enumerate(dates)}

    # 1) 对齐时间序列
    emo = [5.0] * days                       # 情绪分 0-10，正面为高
    for r in db.query(EmotionRecord).filter(
            EmotionRecord.user_id == user_id,
            EmotionRecord.record_date >= dates[0]).all():
        if r.record_date in idx:
            emo[idx[r.record_date]] = r.intensity if r.emotion in ("happy","excited","calm") \
                                      else (10 - r.intensity)

    habits = db.query(Habit).filter_by(user_id=user_id, status=1).all()
    pairs = []
    for h in habits:
        done = [0.0] * days
        for it in db.query(CheckinItem).filter(
                CheckinItem.user_id == user_id, CheckinItem.habit_id == h.id,
                CheckinItem.created_at >= dates[0]).all():
            d = it.created_at.date()
            if d in idx and it.status == "done":
                done[idx[d]] = 1.0

        if sum(done) < 3:
            continue                          # 样本太少不做结论（避免假相关）

        # 同期相关 + 滞后 1 天相关（今天打卡 → 明天情绪）
        r_same = pearson(done, emo)
        r_lag = pearson(done[:-1], emo[1:])

        # 只保留有意义的（|r| ≥ 0.3），weak/strong 分级
        best = r_lag if abs(r_lag) > abs(r_same) else r_same
        if abs(best) >= 0.3:
            pairs.append({
                "habit_id": h.id, "habit_name": h.name,
                "r": best, "r_same": r_same, "r_lag": r_lag,
                "direction": "positive" if best > 0 else "negative",
                "strength": "strong" if abs(best) >= 0.6 else "moderate",
                "lag_days": 0 if best == r_same else 1,
                "n": days,
            })

    # 2) 摆烂 → 情绪 的条件概率（比相关系数更好理解）
    slack_flags, bad_mood = [], []
    for d in dates:
        s = db.query(CheckinItem).filter(
            CheckinItem.user_id == user_id, CheckinItem.status == "slacking",
            func.date(CheckinItem.created_at) == d).count()
        slack_flags.append(1 if s else 0)
        bad_mood.append(1 if emo[idx[d]] < 5 else 0)

    p_bad_given_slack = (sum(1 for i in range(days)
                            if i > 0 and slack_flags[i-1] and bad_mood[i])
                         / max(sum(slack_flags[:-1]), 1))
    p_bad_base = sum(bad_mood) / days

    # 3) 交给大模型翻译成"人话"（结果缓存进 insight_reports，不重复烧 token）
    insight = AIService.chat_non_stream([
        {"role": "system", "content": CAUSALITY_PROMPT},   # 要求：只基于给定数字，禁止编造与鸡汤
        {"role": "user", "content": json.dumps({
            "pairs": pairs,
            "p_bad_given_slack": round(p_bad_given_slack, 2),
            "p_bad_base": round(p_bad_base, 2),
            "lift": round(p_bad_given_slack / p_bad_base, 2) if p_bad_base else 0,
        }, ensure_ascii=False)},
    ])
    return {"pairs": pairs, "probability": {...}, "insight": insight, "sample_days": days}
```

**`CAUSALITY_PROMPT` 关键约束**

```
你是数据分析师，不是鸡汤作者。
1. 只能使用我提供的数字，禁止编造任何数据
2. 禁止出现"加油""相信自己""你可以的"等鼓励语
3. 每条洞察必须写成"因为…所以…"的因果句，并附上支撑数字
4. 如果相关性弱（|r|<0.4），必须主动说明"这只是关联，不是因果"
5. 语言平实，像体检报告，不像励志演讲
```

## 5. 项目创新点（可直接复制到课程报告 / 竞赛文档）

### 5.1 市场痛点（可直接引用）

> 当前主流习惯养成类产品普遍存在四重割裂：
> **其一，数据割裂。** AI 聊天与打卡行为数据分属两套系统，AI 只读取对话历史，对用户"连续三天没早起"这类关键行为一无所知，导致对话空洞、缺乏针对性，用户很快感到"它并不真的懂我"。
> **其二，评价体系单一。** 打卡只支持"完成"这一种正向记录，摆烂、拖延、失控无处安放。用户一旦失败只能选择沉默或放弃，产品实际上在制造心理负罪感，而这恰恰是习惯养成最大的流失原因。
> **其三，人格固化。** AI 人设由用户一次性选定后便固定不变，无法随用户真实状态浮动——在用户情绪崩溃时仍用元气语调鼓励，在用户持续摆烂时依然温柔安慰，情感错位严重。
> **其四，激励物空心化。** 宝物、徽章等激励物多为纯虚拟装饰，与用户的真实经历无关，无法承载情感价值，导致"收集"沦为无意义的数值游戏，难以形成留存黏性。

### 5.2 本项目的差异化设计（可直接引用）

> 针对上述痛点，本项目提出并实现了一套**"行为—情绪—人格—叙事"四层耦合**的 AI 习惯养成架构，与同类产品形成明确差异：
>
> **① 动态人格自适应，而非手动选择。** 后端在每次调用大模型前，读取用户近 7 天打卡完成率、连续摆烂天数、情绪强度与危机等级，通过可解释的优先级规则（危机 > 情绪 > 摆烂 > 成就 > 默认）自动决策当前应采取「清醒督促」「共情陪伴」还是「鼓励玩伴」人格，并动态拼接 system prompt。全程无需训练模型、无需用户手动选择，且每次切换都留有证据链（`ai_persona_logs`），可解释、可追溯。特别地，本项目将**"情绪优先于督促"设为伦理红线**——当用户连续低落时，系统绝不施加任何督促压力。
>
> **② 习惯记忆锚点，让 AI 真正"记得"。** 打通对话记忆与行为数据，将打卡记录、情绪诱因、作息模式沉淀为带权重的行为标签与长期记忆条目，在话题相关时条件注入 prompt。AI 因此能主动说出"你最近 7 天有 4 天晚睡"，而不是泛泛地询问"发生什么了"。同时通过权重、过期时间与命中计数实现记忆的冷热淘汰，避免陈旧记忆干扰对话。
>
> **③ 摆烂反向打卡，把失败变成数据。** 首创「完成 / 摆烂 / 暂停」三态打卡模型：用户可主动记录拖延与失控，填写诱因标签；AI 只做复盘、绝不指责，并遵循"不扣经验、不清零成就、不显示红色"的免愧疚原则。系统进一步将摆烂诱因聚合为 Top 榜与星期分布，让用户看见自己的模式而非缺点。
>
> **④ 分层耦合 MDP 自适应目标引擎。** 在既有难度规则之上叠加一层马尔可夫决策过程：以近 7 天完成率为状态、以升降级为动作、以完成与摆烂为奖励，通过 Q-learning 学习用户个体的最优目标难度，并设置三重安全护栏（连续摆烂强制降级、连续完成方可升级、7 天调整冷却）与 25% 幅度限制，在"推动成长"与"防止挫败"之间取得平衡。降级不被视为失败，而被明确表述为"保护你的持续参与"。
>
> **⑤ 双维度情绪宝物系统。** 将宝物从纯装饰升级为经历的载体，明确分为「成就宝物」（来自坚持）与「情绪宝物」（来自摆烂、崩溃、内耗等负面经历），二者在展示权重上完全平等。每件宝物可解锁专属 AI 对话片段，宝物组合还能触发专属剧情，使"崩溃"本身成为可收藏、可回溯、有叙事价值的资产——这在同类产品中极为罕见。
>
> **⑥ 习惯—情绪双维度匿名广场。** 动态可挂载 `#摆烂记录` `#坚持打卡` 等习惯标签，支持"只看摆烂 / 只看坚持"筛选。产品设计上有意采用**时间倒序而非热度排序**、成就与摆烂等权展示、摆烂页配"这里不评比，只看真实"的氛围文案，从机制层面抑制内卷攀比，鼓励真实表达。
>
> **⑦ AI 习惯互助配对。** 基于用户习惯弱点标签与当前状态计算相似度，匹配"正在同一个坑里"的伙伴，而非匹配优秀榜样。AI 作为中间人，在一方连续摆烂时自动介入调解，把监督关系转化为低压力的陪伴关系；配对默认关闭、双向脱敏、可无理由解除且不扣分。
>
> **⑧ 习惯—情绪因果仪表盘。** 在关联统计之上引入皮尔逊相关系数、滞后相关与条件概率提升度（lift），例如量化"连续熬夜后次日情绪分平均下降 2.3 分""摆烂次日情绪低落概率 71%，显著高于 38% 的基线"，再由大模型将数字翻译为平实的人话洞察。系统明确标注"这是关联而非因果"，并禁止输出鼓励语，以体检报告式的克制替代通用鸡汤。
>
> 综上，本项目并未停留在"AI + 打卡"的功能叠加，而是以**真实数据驱动 AI 人格、以负面经历重构激励体系、以可解释算法替代统一施压**为三条主线，构建出一套更懂人、也更容得下失败的习惯养成系统。

> **"一个允许你摆烂的习惯 App——你的崩溃也会被认真收藏。"**

```python
# server/app/services/calendar_service.py

class CalendarService:
    """日历视图核心是按月聚合，单月最多 38 个 cell × 多源数据。
    用一次 SQL 把所有数据拉到内存，避免 N+1。"""

    @staticmethod
    def month_view(db, user_id, year: int, month: int) -> dict:
        first = date(year, month, 1)
        if month == 12:
            last = date(year + 1, 1, 1) - timedelta(days=1)
        else:
            last = date(year, month + 1, 1) - timedelta(days=1)

        # ---- 1. 当月所有日期的情绪（取每天最后一条为主）----
        emo_rows = db.query(EmotionRecord).filter(
            EmotionRecord.user_id == user_id,
            EmotionRecord.record_date >= first,
            EmotionRecord.record_date <= last,
        ).order_by(EmotionRecord.record_date, EmotionRecord.id.desc()).all()
        # 同一天多条时按 id desc 排序，第一条即为最后一条
        emo_by_date = {}
        seen = set()
        for r in emo_rows:
            if r.record_date in seen:
                continue
            seen.add(r.record_date)
            emo_by_date[r.record_date] = {"emotion": r.emotion, "intensity": r.intensity}

        # ---- 2. 当月所有打卡主表（一行/天）----
        chk_rows = db.query(Checkin).filter(
            Checkin.user_id == user_id,
            Checkin.checkin_date >= first,
            Checkin.checkin_date <= last,
        ).all()
        chk_by_date = {c.checkin_date: c for c in chk_rows}

        # ---- 3. 当月所有打卡明细 ----
        items_rows = db.query(CheckinItem).filter(
            CheckinItem.user_id == user_id,
            CheckinItem.created_at >= datetime.combine(first, datetime.min.time()),
            CheckinItem.created_at < datetime.combine(last + timedelta(days=1), datetime.min.time()),
        ).all()
        items_by_date = {}
        for it in items_rows:
            d = it.created_at.date()
            items_by_date.setdefault(d, []).append(it)

        # ---- 4. 当月所有日记 ----
        notes_rows = db.query(DailyNote).filter(
            DailyNote.user_id == user_id,
            DailyNote.note_date >= first,
            DailyNote.note_date <= last,
        ).all()
        notes_by_date = {}
        for n in notes_rows:
            notes_by_date.setdefault(n.note_date, []).append(n)

        # ---- 5. 当月解锁的宝物（点缀日历）----
        treasure_rows = db.query(UserTreasure, TreasureDef).join(
            TreasureDef, TreasureDef.id == UserTreasure.treasure_id
        ).filter(
            UserTreasure.user_id == user_id,
            func.date(UserTreasure.unlocked_at) >= first,
            func.date(UserTreasure.unlocked_at) <= last,
        ).all()
        treasures_by_date = {}
        for ut, td in treasure_rows:
            d = ut.unlocked_at.date()
            treasures_by_date.setdefault(d, []).append({"name": td.name, "icon": td.icon})

        # ---- 6. 逐日组装（填充空白日）----
        days = []
        for i in range((last - first).days + 1):
            d = first + timedelta(days=i)
            emo = emo_by_date.get(d)
            chk = chk_by_date.get(d)
            days.append({
                "date": d.isoformat(),
                "has_note": bool(notes_by_date.get(d)),
                "mood_summary": emo,
                "checkin": {
                    "done_count": chk.completed_count if chk else 0,
                    "total": chk.total_habits if chk else 0,
                    "rate": float(chk.completion_rate) if chk else 0,
                    "slack_count": chk.slack_count if chk else 0,
                },
                "treasures_unlocked": treasures_by_date.get(d, []),
            })

        # ---- 7. 月度统计 ----
        month_stats = {
            "notes_count": sum(1 for d in days if d["has_note"]),
            "slack_total": sum(d["checkin"]["slack_count"] for d in days),
            "checkin_days": sum(1 for d in days if d["checkin"]["done_count"] > 0),
            "longest_streak": _calc_longest_streak(chk_rows),
            "treasures_count": sum(len(d["treasures_unlocked"]) for d in days),
        }
        return {"days": days, "month_stats": month_stats}
```

```python
        return _fallback_recommendations(audio_map, limit)

    primary = row.primary_type
    cands = PERSONALITY_TO_MEDITATION.get(primary, [])
    out = []
    for session_type, reason in cands[:limit]:
        audios = [a for a in audio_map.get(session_type, [])
                  if (not a.recommended_for_personality)
                  or primary in json.loads(a.recommended_for_personality)]
        if audios:
            a = audios[0]
            out.append({
                "audio_id": a.id, "title": a.title, "audio_url": a.audio_url,
                "cover_url": a.cover_url, "duration_min": a.duration_min,
                "session_type": session_type,
                "narrator": a.narrator, "reason": reason,
            })
    # 落到 user_settings.mindfulness_recommended_type（下次离线推荐）
    if out:
        _upsert_user_setting(db, user_id, mindfulness_recommended_type=out[0]["session_type"])
    return out
```

### 6.6 风险与规避

| 风险 | 规避 |
|---|---|
| 树洞内容被纳入 AI 长期记忆后影响正常聊天人格 | `AIUserMemory.is_long_term=1` 但用 `tags=["treehole"]` 区分，调人格时**只引用 1-2 条最近树洞摘要**，不全文拼入 prompt |
| 日历聚合 SQL 性能问题（一次查多表） | `month_view` 走 5 次单表查询 + Python 端聚合，**避免 JOIN 大表**；后续如用户量大可加 `calendar_marks` 物化视图 |
| 性格测评多版本结果导致混乱 | 用 `is_active` 标志位 + 提交时先 UPDATE 旧记录再 INSERT 新记录；前端默认只展示 active=1 的结果 |
| localStorage 与云端状态不一致 | 默认不同步；"同步到云端"开关是单向开启，用户主动确认；冲突时以**云端为准**（登录态下 GET 云端覆盖本地） |
| 树洞被用于危机自伤表达 | `description` 内置关键词扫描；命中时自动跳危机服务并显示本地求助资源（复用 `crisis_*` 模块与 `app_configs` 配置的求助热线） |
| 推送模板被滥用为垃圾内容 | 输入校验：≤ 200 字、不允许 HTML 标签、占位符只允许白名单 `{habit_name}/{streak}/{date}/{mood}`；保存前先在前端正则校验 |
| 设置页开关被双写打乱原有 localStorage 行为 | "同步到云端"开关**默认关闭**；开启后才走双写；任何失败都不影响本地读写 |

## 7. 功能 11 · MBTI + 大五人格 专业测评模块（v1.2 增量）

> **本模块定位**：在现有「4 类人格（thinker/feeler/doer/dreamer）」之上**叠加**一套基于经典心理量表的科学测评，**不删除**原 4 类体系。
> **两条铁律**：
> 1. 量表**拒绝娱乐化**：禁止照搬"测测你是《哈利波特》谁"这类题目；题目用校园学习/社交/情绪场景包装；正反向题混合；Likert 5 点量表（1=非常不同意 … 5=非常同意）。
> 2. 报告**后端算前端展**：分数、字母、雷达、推荐全部由 `personality_service` 计算生成；前端 `pages/personality-v2/` 只做答题 + 渲染 + 引导"应用到 AI 伙伴"。
> 3. AI 联动走 `ai_user_memories.memory_type='personality'` 长期记忆通道（已在 v1.1 第 0.1 行确认 `is_long_term=1` 字段存在），chat / treehole / mindfulness prompt 注入复用现有 `ai_service.should_inject_*` 钩子，**不动现有 prompt 主干**。

### 7.0 模块定位：与现有 4 类人格体系的关系

| 维度 | 现有 v1.1 体系（保留） | 本轮 v1.2 新增 |
|---|---|---|
| 量表性质 | 8 题左右单选情景题（娱乐化） | MBTI 50 题 + 大五 50 题（专业量表） |
| 数据存储 | `user_personality_results`（主型 4 选 1） | 5 张新表（见 §7.1） |
| 测评入口 | `pages/personality-quiz/` | 新增 `pages/personality-v2/`（老页面保留为兜底） |

| 报告维度 | 一段文字 + AI 伙伴推荐 | 16 种 MBTI × 4 字母 × 3 变体 + 大五 5 维雷达 + 优势短板 + 习惯适配 + AI 人设 + 成长建议 |

#### 7.3.1 MBTI 评分算法（含反向计分 + 归一化）

```python
# server/app/services/personality_service.py

def score_mbti(answers: List[AnswerIn], version: str) -> MBTIResult:
    """answers 已经过校验：长度=50，value∈[1,5]，每题 q_id 在 version 题库内"""
    buckets = {"EI": {"E": 0.0, "I": 0.0, "n": 0},
               "SN": {"S": 0.0, "N": 0.0, "n": 0},
               "TF": {"T": 0.0, "F": 0.0, "n": 0},
               "JP": {"J": 0.0, "P": 0.0, "n": 0}}
    qmap = {q.id: q for q in load_active_questions(scale="mbti", version=version)}

    for ans in answers:
        q = qmap[ans.q_id]
        # Likert 1-5 → 反向题翻转；正向题保持
        score = float(ans.value) if q.direction == "positive" else 6.0 - ans.value
        b = buckets[q.dimension]
        b[q.pole] += score  # pole='E' 累到 E，pole='I' 累到 I
        b["n"] += 1

    # 各维度字母：分高的极胜出
    letters = {}
    raw_diff = {}  # 净分绝对值，用于判断偏好强度
    norm_scores = {}  # 0-100 归一化
    for dim, sides in buckets.items():
        left, right = list(sides.keys())[0], list(sides.keys())[1]
        l_sum, r_sum = sides[left], sides[right]
        diff = abs(l_sum - r_sum)
        max_diff = sides["n"] * 5  # 理论最大差
        norm = 50 + ((l_sum - r_sum) / max_diff) * 50  # 50 中位
        norm = max(0.0, min(100.0, norm))
        letters[dim] = left if l_sum >= r_sum else right
        raw_diff[dim] = diff
        norm_scores[dim] = round(norm, 1)

    mbti_type = letters["EI"] + letters["SN"] + letters["TF"] + letters["JP"]
    return MBTIResult(
        ei_letter=letters["EI"], sn_letter=letters["SN"],
        tf_letter=letters["TF"], jp_letter=letters["JP"],
        mbti_type=mbti_type,
        ei_score=norm_scores["EI"], sn_score=norm_scores["SN"],
        tf_score=norm_scores["TF"], jp_score=norm_scores["JP"],
        dimension_scores=norm_scores,
        raw_diff=raw_diff,
    )
```

#### 7.3.2 大五 OCEAN 评分

```python
def score_bigfive(answers: List[AnswerIn], version: str) -> BigFiveResult:
    trait_sums = {"O": 0.0, "C": 0.0, "E": 0.0, "A": 0.0, "N": 0.0}
    counts = {"O": 0, "C": 0, "E": 0, "A": 0, "N": 0}
    qmap = {q.id: q for q in load_active_questions(scale="bigfive", version=version)}

    for ans in answers:
        q = qmap[ans.q_id]
        score = float(ans.value) if q.direction == "positive" else 6.0 - ans.value
        trait_sums[q.trait] += score
        counts[q.trait] += 1

    # 每维度 10 题 × 1-5 分，平均分 (1-5) → 归一化 0-100
    ocean = {}
    for t in "OCEAN":
        avg = trait_sums[t] / counts[t] if counts[t] else 3.0
        ocean[t] = round((avg - 1.0) / 4.0 * 100.0, 1)
    # 三档分桶：< 40 low, 40-70 mid, > 70 high（标准差由样本决定，先用百分位）
    levels = {t: ("low" if v < 40 else "mid" if v < 70 else "high") for t, v in ocean.items()}
    return BigFiveResult(ocean_scores=ocean, ocean_levels=levels, raw=trait_sums, n_per_trait=counts)
```

#### 7.3.3 `-T / -A / -C` 变体判定（参考 16Personalities 的 Assertive/Turbulent 分流）

```python
def determine_variant(mbti_type: str, ocean: dict, jp_letter: str) -> str:
    """在经典 MBTI 之外叠加一个变体提示，更贴近用户的真实行为节奏。
    -A (Assertive 自信型)：神经质低 + 自我肯定强
    -T (Turbulent 湍流型)：神经质高 + 完美主义倾向
    -C (Constructive 建设型)：尽责性高 + 行动导向
    """
    n = ocean.get("N", 50)
    c = ocean.get("C", 50)
    if n >= 60:
        return "-T"  # 高神经质 → 湍流
    if c >= 65 and jp_letter == "J":
        return "-A"  # 高尽责 + 判断型 → 自信建设
    return "-C"    # 默认：建设型
```

```python
# server/app/services/persona_service.py（在原有 evaluate() 上叠加分支）

def evaluate(db, user_id: int) -> dict:
    """返回给 chat/treehole/mindfulness 的性格画像。
    优先级：MBTI (is_active=1) > 大五 (is_active=1) > 4 类兼容层 > 默认"""

    mbti = db.query(UserPersonalityMBTIResult).filter_by(user_id=user_id, is_active=1).first()
    if mbti:
        type_info = db.query(PersonalityDimensionType).filter_by(
            scale_type="mbti", code=mbti.mbti_type, variant=mbti.variant
        ).first()
        return {
            "source": "mbti",
            "mbti_type": mbti.mbti_type,
            "variant": mbti.variant,
            "primary_4type": MBTI_TO_4TYPE.get(mbti.mbti_type, "thinker"),
            "tone": _mbti_to_tone(mbti.mbti_type),
            "prompt_addon": type_info.ai_persona_hint if type_info else "",
            "growth_caveat": type_info.growth_suggestion if type_info else "",
        }

    # 兜底：4 类兼容层（v1.1 用户）
    legacy = db.query(UserPersonalityResult).filter_by(user_id=user_id, is_active=1).first()
    if legacy:
        return {"source": "legacy_4type", "primary_4type": legacy.primary_type, ...}

    return {"source": "default", "primary_4type": "thinker", "tone": "rational", ...}


def _mbti_to_tone(mbti_type: str) -> str:
    """把 MBTI 映射成 4 种 AI 人设 tone，复用 v1.1 TREEHOLE_PERSONA 的 tone_default"""
    tf = mbti_type[2]  # T 或 F
    return "rational" if tf == "T" else "gentle"  # 其他维度留给 PersonaService 按场景叠加
```

> **关键不破坏**：原 `evaluate()` 返回的字段（`primary_4type`、`tone`、`prompt_addon`）**字段名/语义都不变**，v1.1 全部调用方零修改。

#### 7.3.8 `treehole_service` 与 `mindfulness_service` 自动吃到 MBTI（零代码改动）

由于 `PersonaService.evaluate()` 已经升级且返回字段不变：
- `treehole_service.create()` 调 `PersonaService.evaluate(db, user.id).get('prompt_addon')` → 拿到 MBTI 的 `ai_persona_hint`，自动比 v1.1 的 4 类话术**更精准**
- `mindfulness_service.recommend_for_user()` 同理，MBTI 维度的 `habit_fit` 优先于 4 类映射

**这就是"不推翻"的工程含义**：上游字段不变，下游计算更精准。

#### 7.4.5 可选层：AI 解读（智谱二次润色）

如果产品想让报告文案更有人味，可让后端在算分后**异步**调智谱：

```python
def ai_enhance_report(mbti_type, ocean_scores, dimension_scores, type_info):
    """可选：把骨架文案喂给智谱润色，输出更像心理学科班编辑写的"""
    if not settings.AI_REPORT_ENHANCE_ENABLED:  # 配置开关，默认关闭
        return {"summary": type_info.one_liner, "growth": type_info.growth_suggestion}
    prompt = f"""你是校园心理学科班编辑。基于 MBTI={mbti_type} 大五={json.dumps(ocean_scores)}
重写一段'成长建议'（100 字内），不要给出临床诊断，不要用'你应该'命令句。
原建议：{type_info.growth_suggestion}"""
    summary = ai_service.chat_non_stream([{"role": "system", "content": prompt}])
    return {"summary": summary.content}
```

- 默认**关闭**（`AI_REPORT_ENHANCE_ENABLED=false`），走静态字典文案，省 token 成本；
- 开启后也只跑**一次**（结果存 `user_personality_*_results.ai_enhanced_text`），避免每次看报告都花钱；

## 附录 B · 关键风险与规避

| 风险 | 规避措施 |
|---|---|
| 摆烂打卡被用户当成"免罪符"，打卡量暴跌 | 摆烂不产出 XP、不进完成率分子，仅产出情绪宝物；`completion_rate` 计算剔除摆烂与暂停 |
| 自动调目标让用户感到失控 | 全部展示 `reason` 文案；提供一键关闭；调整幅度 ≤25%；7 天冷却 |
| 情绪宝物被解读为"奖励摆烂" | 文案统一为"记录"而非"奖励"；宝物名偏向承载感（雨瓶、金缮、夜灯）而非荣誉感 |
| AI 洞察给出虚假因果 | 强制 `|r| ≥ 0.3` 才展示；样本 < 7 天不出结论；prompt 硬性要求声明"关联非因果" |
| 匹配功能引发隐私顾虑 | 默认关闭、双向脱敏、仅暴露状态不暴露内容、可无理由解除 |
| 新增表破坏现有 `init_db.py` 建表 | 所有新字段给全默认值（`add_server_defaults` 要求）；新表通过同一 `Base.metadata` 注册 |
| FastAPI 8000 改动不生效 | `APP_DEBUG=false` 时需**手动重启** 8000 服务（Flask 5001 自动重载） |
| **v1.2 题库 pole/direction 被前端白嫖导致绕过计分** | **`/personality/questions` 不下发 `pole` 与 `direction` 字段**，仅返回题目与 A/B 选项；前端无论如何点击，原始分数由后端按 pole/direction 计算 |
| **v1.2 大五分数被用户误解为临床诊断** | **报告页底部固定 disclaimer 文案**：「本测评基于 IPIP 风格自建 OCEAN 简版量表，仅供自我认知参考，不属于临床心理诊断」；OCEAN 维度解读语言克制，禁止出现「你有 XX 症」类表述；disclaimer 文案走 `app_configs` 可运营改 |
| **v1.2 历史测评来回切换导致 AI 人设突变** | **`/personality/mbti/activate/{id}` 切换时，AI 长期记忆同步更新**，新画像覆盖旧画像；同时在 chat 历史里打 `tag=personality_switch`，前端聊天面板顶部弹一次轻提示「你的人设画像已切换为 ENTJ，AI 伙伴会按新画像回应你」 |
| **v1.2 `completion_seconds < 60`（过快）污染历史** | **标记 `flag="too_fast"`** 并在报告页顶部显示「检测到完成时间较短，部分答案可能需要重新考量」；**不强制重测**，由用户决定 |
| **v1.2 AI 解读调智谱的 token 成本与延迟** | **`AI_REPORT_ENHANCE_ENABLED` 默认 `false`**，跑静态字典文案；启用时也只跑**一次**结果存 `ai_enhanced_text`，避免每次看报告都花钱；异步执行不阻塞报告页加载 |
| **v1.2 老 4 类用户（v1.1 测过）迁移断层** | **首次 v1.2 测评完成后，自动反向写兼容层 `user_personality_results`**（4 类）；v1.1 测过的老用户**不影响**，他们的 4 类结果继续生效，新测后才覆盖；前端不显示"升级到 v1.2"提示，避免催促 |
| **v1.2 答题中途退出污染历史** | **答题过程中只写本地 `setStorageSync('p2_answers_*')`**，**不调 submit 接口**；只有用户点"提交"才落库；中途退出 0 副作用 |
| **v1.2 历史超 5 条不被告知** | **超出 5 条自动归档到 `user_personality_*_archive` 表**，前端历史页固定显示「仅保留最近 5 条」；管理员可查 archive 表做数据回查 |
| **v1.2 个性画像分享图被用于社交攀比** | **分享图固定脱敏**：去除 user_id / 昵称 / 学校标签；文案统一为「我的画像」而不是「我比你更 XX」；不允许比较功能（v1.2 不做"看看你俩合不合"） |
| **v1.2 大五雷达图配色让用户误以为分数越高越好** | **每维度下方明示"高/中/低"水平**（不是单纯分数），并写「高神经质是正常的焦虑水平，不代表心理不健康」类解释，避免分数焦虑 |
